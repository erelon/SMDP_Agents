\begin{center}
  {\Large\bfseries CRRA Utility-Transformed SMDP R-Learning}\\[-2pt]
  {\small Pseudocode corresponding to \texttt{agents/risky\_reward\_r.py}}
\end{center}

For a positive local reward rate $x=r/t>0$, define CRRA utility
\[
\CRRA{\theta}{x}=
\begin{cases}
\dfrac{x^{1-\theta}-1}{1-\theta}, & \theta\neq 1,\\[5pt]
\log x, & \theta=1,
\end{cases}
\qquad
M_\theta^{-1}(y)=
\begin{cases}
\bigl(1+(1-\theta)y\bigr)^{1/(1-\theta)}, & \theta\neq1,\\[3pt]
e^y, & \theta=1.
\end{cases}
\]
Here $\theta>0$ is risk-averse, $\theta=0$ is risk-neutral, and $\theta<0$
is risk-seeking.  For $p\in\mathbb{R}$, the duration-weighted power mean is
\[
\PMean{p}{x_{1:K};t_{1:K}}=
\begin{cases}
\left(\dfrac{\sum_{i=1}^{K}t_i x_i^p}{\sum_{i=1}^{K}t_i}\right)^{1/p},
  & p\neq0,\\[8pt]
\exp\!\left(\dfrac{\sum_{i=1}^{K}t_i\log x_i}{\sum_{i=1}^{K}t_i}\right),
  & p=0.
\end{cases}
\]
The exact relationship, with $p=1-\theta$, is
\[
\boxed{
M_\theta^{-1}\!\left(
  \frac{\sum_{i=1}^{K}t_i\CRRA{\theta}{x_i}}{\sum_{i=1}^{K}t_i}
\right)
=\PMean{1-\theta}{x_{1:K};t_{1:K}}.}
\]

\begin{algorithm}[H]
\small
\caption{CRRA utility-transformed SMDP R-learning}
\KwIn{$\theta$, Q-learning rate $\alpha$, exploration rate $\varepsilon$,
      action set $\mathcal A$, and whether to use the $\rho$ trick}
$p\leftarrow1-\theta$; $Q(s,a)\leftarrow0$; $U\leftarrow0$;
$T\leftarrow0$; $\rho\leftarrow M_\theta^{-1}(0)=1$\;
\ForEach{transition $(s,a,r,t,s')$}{
  Require $t>0$ and set $x\leftarrow r/t>0$\;
  $a^*\leftarrow\arg\max_{b\in\mathcal A}Q(s',b)$\;
  $g_\theta\leftarrow t\,[\CRRA{\theta}{x}-\CRRA{\theta}{\rho}]$\;
  $\delta\leftarrow g_\theta+Q(s',a^*)-Q(s,a)$\;
  $Q(s,a)\leftarrow Q(s,a)+\alpha\delta$\;
  \If{the $\rho$ trick is disabled, or $a$ was greedy in $s$}{
    $U\leftarrow U+t\CRRA{\theta}{x}$; $T\leftarrow T+t$\;
    $\bar u\leftarrow U/T$; $\rho\leftarrow M_\theta^{-1}(\bar u)
      =\PMean{p}{x_{1:K};t_{1:K}}$\;
  }
  Select the next action $\varepsilon$-greedily from $Q$\;
}
\end{algorithm}

For $\theta=0$, $M_0(x)=x-1$, hence
$t[M_0(r/t)-M_0(\rho)]=r-\rho t$; the TD target therefore reduces exactly
to ordinary continuous R-learning.
